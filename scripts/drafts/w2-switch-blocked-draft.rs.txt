/// Mounts a store at a **directory** mount point (the letter pool is exhausted by the other cases, and a directory is the other supported shape).
fn mount_at_directory(label: &str) -> (PathBuf, MountChild, PathBuf, PathBuf) {
    let store = temp_dir(label);
    Command::new(binary())
        .arg("init")
        .arg(&store)
        .output()
        .expect("init");
    let mount_point = store.join("mnt");
    let child = MountChild(
        Command::new(binary())
            .arg("--store")
            .arg(&store)
            .arg("mount")
            .arg(&mount_point)
            .arg("--workspace")
            .arg("main")
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .spawn()
            .expect("spawn mount"),
    );
    // **Ready means the record exists**, not that the directory does: a directory mount point is on the host volume and exists **before** the volume is attached
    // (the A6d lesson). Waiting on `metadata` here let the first `checkout` run before the mount was even recorded.
    let record = fs_cli::mounts_dir().join(format!("{}.mount", fs_cli::record_name(&mount_point)));
    let started = Instant::now();
    while started.elapsed() < Duration::from_secs(15)
        && !std::fs::read_to_string(&record).unwrap_or_default().contains("pipe=")
    {
        std::thread::sleep(Duration::from_millis(50));
    }
    assert!(
        std::fs::read_to_string(&record).unwrap_or_default().contains("pipe="),
        "cell setup: the mount must be recorded and listening"
    );
    let root = mount_point.clone();
    (store, child, root, mount_point)
}

/// **W2** — an open **file** handle blocks a switch, the mount keeps serving the old workspace, and a later attempt succeeds.
#[test]
fn an_open_file_handle_blocks_the_switch() {
    common::strict_gate();
    let (store, _child, root, mount_point) = mount_at_directory("switch-blocked");
    Command::new(binary())
        .arg("--store")
        .arg(&store)
        .args(["branch", "feat", "--from", "main"])
        .output()
        .expect("branch feat");
    // **A branch is not a workspace** (A3b): `checkout <b>` creates the workspace row on first use, and that first call does **not** switch. Doing it here, from
    // outside the mount, is what makes the checkout below the switch rather than a creation.
    let (created, out, err) = cli_code(&store, &store, &["checkout", "feat"]);
    assert_eq!(created, 0, "cell setup: the feat workspace row must exist: {out}{err}");
    assert!(
        out.contains("on feat"),
        "cell setup: the first checkout creates the workspace, got {out:?}"
    );
    let file = root.join("f");
    std::fs::write(&file, b"main content").expect("write");

    let held = std::fs::File::open(&file).expect("hold the file open");
    let (code, out, err) = cli_code(&store, &root, &["checkout", "feat"]);
    println!("W2 reading: blocked -> code={code} stderr={:?}", err.trim());
    assert_eq!(code, 4, "cell W2: an open file handle must make the switch busy (4), got {code}: {out}{err}");
    assert!(err.contains("1 file"), "cell W2: the refusal names the file count, got {err:?}");

    // The mount still serves `main`, and it can be opened: a refusal that left the gate closed would hang here.
    let still = std::fs::read(&file).expect("the mount still reads its file");
    assert_eq!(still, b"main content", "cell W2: the view did not move");
    drop(held);

    let (code, out, err) = cli_code(&store, &root, &["checkout", "feat"]);
    println!("W2 reading: after closing -> code={code} stdout={:?}", out.trim());
    assert_eq!(code, 0, "cell W2: with no handle held the switch must succeed: {out}{err}");
    let (code, out, _) = cli_code(&store, &root, &["status"]);
    println!("W2 reading: status after the switch -> code={code} stdout={:?}", out.trim());
    assert_eq!(code, 0, "cell W2: status must answer");
    assert!(
        out.contains("workspace: feat") || out.contains("branch:    feat"),
        "cell W2: the mount must serve feat now, got {out:?}"
    );
    let _ = std::fs::remove_dir_all(&store);
}
