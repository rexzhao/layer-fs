/// **W3** — if the target workspace is held by another process, the switch is **busy** (4) and the mount never stops answering.
///
/// Liveness is asked over the **control pipe**, not through the file system: `metadata(mount_point)` says "invisible" for a whole class of reasons that have nothing
/// to do with the mount being gone (a directory mount point that is briefly unusable, a sharing conflict on the parent), and W3 is about the mount, not about
/// `metadata`.
#[test]
fn a_switch_to_a_held_workspace_is_busy_and_the_mount_keeps_answering() {
    common::strict_gate();
    let (store, _child, root) = mount_at_directory("switch-held");
    Command::new(binary())
        .arg("--store")
        .arg(&store)
        .args(["branch", "feat", "--from", "main"])
        .output()
        .expect("branch feat");
    let (created, out, err) = cli_code(&store, &store, &["checkout", "feat"]);
    assert_eq!(created, 0, "cell setup: the feat workspace row must exist: {out}{err}");

    // Hold `feat` from another process: a second mount of the same workspace, at its own mount point.
    let other_point = store.join("other");
    let mut holder = MountChild(
        Command::new(binary())
            .arg("--store")
            .arg(&store)
            .arg("mount")
            .arg(&other_point)
            .arg("--workspace")
            .arg("feat")
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .spawn()
            .expect("spawn the holder mount"),
    );
    let held_record =
        fs_cli::mounts_dir().join(format!("{}.mount", fs_cli::record_name(&other_point)));
    let started = Instant::now();
    while started.elapsed() < Duration::from_secs(15)
        && !std::fs::read_to_string(&held_record).unwrap_or_default().contains("pipe=")
    {
        std::thread::sleep(Duration::from_millis(50));
    }
    assert!(
        std::fs::read_to_string(&held_record).unwrap_or_default().contains("pipe="),
        "cell setup: the holder mount must be serving feat"
    );
    let pipe = std::fs::read_to_string(fs_cli::mounts_dir().join(format!(
        "{}.mount",
        fs_cli::record_name(&root)
    )))
    .unwrap_or_default()
    .lines()
    .find_map(|line| line.strip_prefix("pipe=").map(str::to_string))
    .unwrap_or_default();
    assert!(!pipe.is_empty(), "cell setup: the mount must have a control pipe");

    // Liveness = **the mount answers**, sampled for the whole attempt.
    let stop = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
    let silent = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
    let samples = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
    let poller = {
        let stop = std::sync::Arc::clone(&stop);
        let silent = std::sync::Arc::clone(&silent);
        let samples = std::sync::Arc::clone(&samples);
        std::thread::spawn(move || {
            while !stop.load(std::sync::atomic::Ordering::SeqCst) {
                samples.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
                let request = fs_cli::pipe::Request {
                    v: fs_cli::pipe::PROTOCOL,
                    op: "hello".to_owned(),
                    args: serde_json::json!({}),
                };
                if fs_cli::pipe_win::call(&pipe, &request, 500).is_err() {
                    silent.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
                }
                std::thread::sleep(Duration::from_millis(10));
            }
        })
    };

    let (code, out, err) = cli_code(&store, &root, &["checkout", "feat"]);
    stop.store(true, std::sync::atomic::Ordering::SeqCst);
    poller.join().expect("the poller finishes");
    let silent = silent.load(std::sync::atomic::Ordering::SeqCst);
    let samples = samples.load(std::sync::atomic::Ordering::SeqCst);
    println!(
        "W3 reading: code={code} samples={samples} silent={silent} stdout={:?} stderr={:?}",
        out.trim(),
        err.trim()
    );
    assert_eq!(
        code, 4,
        "cell W3: a workspace another process holds is busy (4), got {code}: {out}{err}"
    );
    assert_eq!(
        silent, 0,
        "cell W3: the mount must answer `hello` throughout a refused switch ({silent} of {samples} unanswered)"
    );
    let (code, out, _) = cli_code(&store, &root, &["status"]);
    assert_eq!(code, 0, "cell W3: status must answer");
    assert!(
        out.contains("workspace: main"),
        "cell W3: the mount still serves main, got {out:?}"
    );
    let _ = holder.0.kill();
    let _ = holder.0.wait();
    let _ = std::fs::remove_dir_all(&store);
}
